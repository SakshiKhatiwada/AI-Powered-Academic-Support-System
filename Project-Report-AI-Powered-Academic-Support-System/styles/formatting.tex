% ============================================================
% formatting.tex
% TU Report Formatting
% ============================================================

% ------------------------------------------------------------
% Page Size and Margins
% ------------------------------------------------------------
\geometry{
    a4paper,
    top=1in,
    bottom=1in,
    left=1.25in,
    right=1in
}

% ------------------------------------------------------------
% Font
% ------------------------------------------------------------
\setmainfont{Times New Roman}

% ------------------------------------------------------------
% Line Spacing
% ------------------------------------------------------------
\onehalfspacing

% ------------------------------------------------------------
% Paragraph Formatting
% ------------------------------------------------------------
\setlength{\parindent}{0pt}
\setlength{\parskip}{0.5em}

% ------------------------------------------------------------
% Prevent ugly stretched spacing
% ------------------------------------------------------------
\emergencystretch=2em

% ------------------------------------------------------------
% Page Numbering
% Bottom Center
% ------------------------------------------------------------
\pagestyle{fancy}

\fancyhf{}

\fancyfoot[C]{\fontsize{12}{14}\selectfont\thepage}

\renewcommand{\headrulewidth}{0pt}
\renewcommand{\footrulewidth}{0pt}

% ------------------------------------------------------------
% Chapter Heading
% 16 pt Bold
% ------------------------------------------------------------
\titleformat{\chapter}[display]
  {\centering\bfseries\fontsize{16pt}{18pt}\selectfont}
  {\MakeUppercase{CHAPTER \thechapter}}
  {1em}
  {\MakeUppercase}

\titlespacing*{\chapter}
{0pt}
{0pt}
{12pt}

% ------------------------------------------------------------
% Section Heading
% 14 pt Bold
% ------------------------------------------------------------
\titleformat{\section}
{\bfseries\fontsize{14pt}{18pt}\selectfont}
{\thesection}
{1em}
{}

% ------------------------------------------------------------
% Subsection Heading
% 12 pt Bold
% ------------------------------------------------------------
\titleformat{\subsection}
{\bfseries\fontsize{12pt}{15pt}\selectfont}
{\thesubsection}
{1em}
{}

% ------------------------------------------------------------
% Figure Captions
% ------------------------------------------------------------
\captionsetup[figure]{
    position=bottom,
    justification=centering,
    labelfont=bf,
    textfont=bf,
    font=normalsize,
    skip=8pt
}

% ------------------------------------------------------------
% Table Captions
% ------------------------------------------------------------
\captionsetup[table]{
    position=top,
    justification=centering,
    labelfont=bf,
    textfont=bf,
    font=normalsize,
    skip=8pt
}

% ------------------------------------------------------------
% TOC Spacing
% ------------------------------------------------------------
\setlength{\cftbeforechapskip}{8pt}
\setlength{\cftbeforesecskip}{2pt}

% ------------------------------------------------------------
% Hyperlinks
% Clickable but visually plain
% ------------------------------------------------------------
\hypersetup{
    colorlinks=false,
    pdfborder={0 0 0}
}

% ------------------------------------------------------------
% Figure/Table numbering
% Example:
% Figure 3.1
% Table 4.2
% ------------------------------------------------------------
\numberwithin{figure}{chapter}
\numberwithin{table}{chapter}

% ------------------------------------------------------------
% Better page breaks
% ------------------------------------------------------------
\raggedbottom

% ------------------------------------------------------------
% Avoid club/widow lines
% ------------------------------------------------------------
\clubpenalty=10000
\widowpenalty=10000
\displaywidowpenalty=10000

%==================================================
% Lists
%==================================================

\setlist[itemize]{
    itemsep=2pt,
    topsep=4pt,
    parsep=0pt,
    partopsep=0pt
}

\setlist[enumerate]{
    itemsep=2pt,
    topsep=4pt,
    parsep=0pt,
    partopsep=0pt
}

% loft captions:
\renewcommand{\cftfigpresnum}{Figure~}
\renewcommand{\cftfigaftersnum}{: }
\setlength{\cftfignumwidth}{4.5em}

\renewcommand{\cfttabpresnum}{Table~}
\renewcommand{\cfttabaftersnum}{: }
\setlength{\cfttabnumwidth}{4.5em}
